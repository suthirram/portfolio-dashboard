# PD-059: UI_2.0

* **Status**: Slices 0–2 merged; overall backend in progress; slice 3 implementation in progress under the merged [Wave 2a amendment](PD-059-wave2-cyber-entrance.md)
* **Owner**: project owner
* **Goal**: a third interface, UI_2.0, behind a per-account `ui_mode` the
  super admin sets (`classic | glass | ui2`), built page by page from the
  owner's mockup with every existing behaviour kept, plus a portfolio-wide
  overall performance chart backed by a compact daily collection.
* **Architecture**: `users.ui_mode` (derived from `ui_revamp` until the
  migration runs) mirrored to `<html data-ui>`; one switch point in
  `ShellLayout`; all UI_2.0 code under `frontend/src/ui2/`, CSS under
  `:root[data-ui="ui2"]`. Overall worth: `portfolio_overall` written only by
  `SnapshotStore`, EUR→INR stored per day, gold added on read.
* **Tech stack**: Go 1.26, echo + oapi-codegen strict server, mongo-driver
  v1, zap, OTel; React 18 + Vite + TypeScript, Vitest. No new dependencies.
* **Spec**: [PRD-015](../prds/PRD-015-ui-2.md) ·
  [DD-015](../designs/DD-015-ui-2.md) · visual reference
  [PD-059 reference](../visuals/PD-059-ui-2-reference.html) (the owner's handoff
  mockup: single source of truth for layout and style, built 1:1) · shared board
  [PD-059 status](PD-059-ui-2-status.md).

// The account interfaces (PD-059). Classic goes away at the UI_2.0 cutover.
const (
    UiModeClassic = "classic"
    UiModeGlass   = "glass"
    UiModeUi2     = "ui2"
)

// ValidUiMode reports whether mode is one of the interfaces.
func ValidUiMode(mode string) bool {
    return mode == UiModeClassic || mode == UiModeGlass || mode == UiModeUi2
}

// EffectiveUiMode is the account's interface: the stored UiMode, or, for an
// account the PD-059 migration has not stamped yet, glass when the legacy
// UiRevamp switch is on and classic otherwise.
func (u *User) EffectiveUiMode() string {
    if ValidUiMode(u.UiMode) {
        return u.UiMode
    }
    if u.UiRevamp {
        return UiModeGlass
    }
    return UiModeClassic
}
```

func (h *Controller) setUserFlag(ctx context.Context, userIdHex, field string, enabled bool) (allowed, found bool, err error) {
    return h.setUserFields(ctx, userIdHex, bson.M{field: enabled})
}

// setUserFields is the super-admin per-user switch body: load the target in
// the caller's scope, $set fields (and updated_at), log it. allowed=false:
// the caller is not the super admin (403); found=false: no such user (404).
func (h *Controller) setUserFields(ctx context.Context, userIdHex string, fields bson.M) (allowed, found bool, err error) {
    caller, ok := superAdminCaller(ctx)
    if !ok {
        return false, false, nil
    }
    target, found, err := h.loadTargetUser(ctx, caller, userIdHex)
    if err != nil || !found {
        return true, found, err
    }
    logger := h.reqLog(ctx)
    set := bson.M{"updated_at": time.Now()}
    for field, value := range fields {
        set[field] = value
    }
    if err := h.store.Users.Update(ctx, target.ID, set); err != nil {
        logger.Error("user switch update failed", zap.Any("fields", fields), zap.Error(err))
        return true, true, err
    }
    logger.Info("user switch updated", zap.String("target", target.ID.Hex()), zap.Any("fields", fields))
    return true, true, nil
}
```

// AdminSetUserUiMode sets an account's interface (PD-059): classic, glass or
// ui2. ui_revamp is written alongside (glass ⇔ true) so a rolled-back binary
// still shows Glass correctly. Super admin only; may switch their own account.
func (h *Controller) AdminSetUserUiMode(ctx context.Context, request api.AdminSetUserUiModeRequestObject) (api.AdminSetUserUiModeResponseObject, error) {
    if request.Body == nil || !domain.ValidUiMode(string(request.Body.Mode)) {
        return api.AdminSetUserUiMode400JSONResponse{BadRequestJSONResponse: api.BadRequestJSONResponse{Error: new("mode must be classic, glass or ui2")}}, nil
    }
    mode := string(request.Body.Mode)
    allowed, found, err := h.setUserFields(ctx, request.Id, bson.M{"ui_mode": mode, "ui_revamp": mode == domain.UiModeGlass})
    switch {
    case err != nil:
        return nil, err
    case !allowed:
        return api.AdminSetUserUiMode403JSONResponse{ForbiddenJSONResponse: forbiddenMsg("super admin access required")}, nil
    case !found:
        return api.AdminSetUserUiMode404JSONResponse{NotFoundJSONResponse: notFoundUser()}, nil
    }
    return api.AdminSetUserUiMode204Response{}, nil
}

package domain

import (
    "time"

    "go.mongodb.org/mongo-driver/bson/primitive"
)

// FxRate is the EUR→INR rate one day's overall totals are converted with
// (PD-059). EurInr is INR per 1 EUR; nil means no rate is known. FxDate is
// the trading day the rate belongs to (a weekend carries Friday's).
// Carried marks a fallback after a failed fetch, or a missing rate — the
// History page offers to re-fetch those days.
type FxRate struct {
    EurInr  *float64
    FxDate  string
    Carried bool
}

// OverallDay is one portfolio_overall document: one user's stock totals for
// one day, both buckets converted to INR at that day's stored rate. Gold is
// added at read time. StocksInvestedInr/StocksCurrentInr are nil when the EUR
// bucket is non-zero and no rate is known.
type OverallDay struct {
    UserID            primitive.ObjectID `bson:"user_id"`
    Date              time.Time          `bson:"date"`
    EurInr            *float64           `bson:"eur_inr,omitempty"`
    FxDate            string             `bson:"fx_date,omitempty"`
    FxCarried         bool               `bson:"fx_carried"`
    StocksInvestedInr *float64           `bson:"stocks_invested_inr,omitempty"`
    StocksCurrentInr  *float64           `bson:"stocks_current_inr,omitempty"`
    UpdatedAt         time.Time          `bson:"updated_at"`
}

// OverallFromBuckets converts a snapshot's INR and EUR buckets into INR
// totals at eurInr. The legacy USD bucket is ignored, as /history drops it.
// With a non-zero EUR bucket and no rate, both totals are nil: a conversion
// is never guessed.
func OverallFromBuckets(buckets map[string]RegionSnapshot, eurInr *float64) (invested, current *float64) {
    inr, eur := buckets[CurrencyINR], buckets[CurrencyEUR]
    if eur.Invested == 0 && eur.Current == 0 {
        investedInr, currentInr := round2(inr.Invested), round2(inr.Current)
        return &investedInr, &currentInr
    }
    if eurInr == nil || *eurInr <= 0 {
        return nil, nil
    }
    investedInr := round2(inr.Invested + eur.Invested**eurInr)
    currentInr := round2(inr.Current + eur.Current**eurInr)
    return &investedInr, &currentInr
}

// Put writes the day's rate and the totals computed from buckets at that
// rate, creating the document when absent. A nil total is unset, never
// stored as zero.
func (s *OverallStore) Put(ctx context.Context, userId primitive.ObjectID, date time.Time, rate domain.FxRate, buckets map[string]domain.RegionSnapshot) error {
    ctx, cancel := context.WithTimeout(ctx, writeTimeout)
    defer cancel()
    invested, current := domain.OverallFromBuckets(buckets, rate.EurInr)
    set := bson.M{"fx_carried": rate.Carried, "updated_at": time.Now().UTC()}
    unset := bson.M{}
    putOrUnset(set, unset, "eur_inr", rate.EurInr)
    putOrUnset(set, unset, "stocks_invested_inr", invested)
    putOrUnset(set, unset, "stocks_current_inr", current)
    if rate.FxDate != "" {
        set["fx_date"] = rate.FxDate
    } else {
        unset["fx_date"] = ""
    }
    update := bson.M{"$set": set}
    if len(unset) > 0 {
        update["$unset"] = unset
    }
    _, err := s.col.UpdateOne(ctx, overallFilter(userId, date), update, options.Update().SetUpsert(true))
    return err
}

func putOrUnset(set, unset bson.M, field string, value *float64) {
    if value != nil {
        set[field] = *value
        return
    }
    unset[field] = ""
}

// RefreshStocks recomputes the day's totals from buckets at the rate the
// document already stores, in one pipeline update, so no read is needed. A
// day with no document yet (a History add or paste for a new date) gets one
// with no rate and fx_carried = true: its totals are known when it holds no
// EUR, and otherwise it is offered for re-fetch on History (DD-015 §2.7).
func (s *OverallStore) RefreshStocks(ctx context.Context, userId primitive.ObjectID, date time.Time, buckets map[string]domain.RegionSnapshot) error {
    ctx, cancel := context.WithTimeout(ctx, writeTimeout)
    defer cancel()
    inr, eur := buckets[domain.CurrencyINR], buckets[domain.CurrencyEUR]
    now := time.Now().UTC()
    upsert := options.Update().SetUpsert(true)
    if eur.Invested == 0 && eur.Current == 0 {
        invested, current := domain.OverallFromBuckets(buckets, nil)
        _, err := s.col.UpdateOne(ctx, overallFilter(userId, date), bson.M{
            "$set":         bson.M{"stocks_invested_inr": *invested, "stocks_current_inr": *current, "updated_at": now},
            "$setOnInsert": bson.M{"fx_carried": true},
        }, upsert)
        return err
    }
    convert := func(inrAmount, eurAmount float64) bson.M {
        return bson.M{"$cond": bson.A{
            bson.M{"$gt": bson.A{bson.M{"$ifNull": bson.A{"$eur_inr", 0}}, 0}},
            bson.M{"$round": bson.A{bson.M{"$add": bson.A{inrAmount, bson.M{"$multiply": bson.A{eurAmount, "$eur_inr"}}}}, 2}},
            "$$REMOVE",
        }}
    }
    pipeline := mongo.Pipeline{{{Key: "$set", Value: bson.M{
        "stocks_invested_inr": convert(inr.Invested, eur.Invested),
        "stocks_current_inr":  convert(inr.Current, eur.Current),
        // A new document has no rate yet: flag it for re-fetch.
        "fx_carried": bson.M{"$ifNull": bson.A{"$fx_carried", true}},
        "updated_at": now,
    }}}}
    _, err := s.col.UpdateOne(ctx, overallFilter(userId, date), pipeline, upsert)
    return err
}

// FillUnrated writes the day's rate and totals only when the day has no
// document or one without a rate (e.g. created by a History add before the
// backfill ran). A document that already has a rate — written live by the
// snapshot job — is never touched: the filter misses it, the upsert's insert
// collides on the unique (user_id, date) index, and that duplicate means
// "already rated". Reports whether it wrote.
func (s *OverallStore) FillUnrated(ctx context.Context, userId primitive.ObjectID, date time.Time, rate domain.FxRate, buckets map[string]domain.RegionSnapshot) (bool, error) {
    ctx, cancel := context.WithTimeout(ctx, writeTimeout)
    defer cancel()
    invested, current := domain.OverallFromBuckets(buckets, rate.EurInr)
    set := bson.M{"fx_carried": rate.Carried, "updated_at": time.Now().UTC()}
    unset := bson.M{}
    putOrUnset(set, unset, "eur_inr", rate.EurInr)
    putOrUnset(set, unset, "stocks_invested_inr", invested)
    putOrUnset(set, unset, "stocks_current_inr", current)
    if rate.FxDate != "" {
        set["fx_date"] = rate.FxDate
    }
    update := bson.M{"$set": set}
    if len(unset) > 0 {
        update["$unset"] = unset
    }
    filter := overallFilter(userId, date)
    filter["eur_inr"] = bson.M{"$exists": false}
    _, err := s.col.UpdateOne(ctx, filter, update, options.Update().SetUpsert(true))
    if mongo.IsDuplicateKeyError(err) {
        return false, nil
    }

}

// List returns the user's documents in [from, to], oldest first.
func (s *OverallStore) List(ctx context.Context, userId primitive.ObjectID, from, to time.Time) ([]domain.OverallDay, error) {
    ctx, cancel := context.WithTimeout(ctx, readTimeout)
    defer cancel()
    filter := bson.M{"user_id": userId, "date": bson.M{"$gte": domain.UTCDate(from), "$lte": domain.UTCDate(to)}}
    cursor, err := s.col.Find(ctx, filter, options.Find().SetSort(bson.D{{Key: "date", Value: 1}}))
    if err != nil {
        return nil, err

    defer func() { _ = cursor.Close(ctx) }()
    var days []domain.OverallDay
    if err := cursor.All(ctx, &days); err != nil {
        return nil, err
    }
    return days, nil
}

// LatestBefore returns the user's newest document dated before date, the

// last close before it (a weekend or holiday, carried with its own fx_date,
// not flagged). No close on or before date is unknown and flagged.
func rateOn(days []string, closes map[string]float64, date string) domain.FxRate {
    index := sort.SearchStrings(days, date)

        rate := closes[date]

    }
    previous := days[index-1]
    rate := closes[previous]
    return domain.FxRate{EurInr: &rate, FxDate: previous}

// rateSources is what a run knows about the day's EUR→INR rate. Live is
// today's quote, fetched only when the run is for the current trading day.
// Historical is the requested day's close, fetched only when the run is for
// a past day (`snapshot --date`), so a past row never gets today's quote.
type rateSources struct {
    Live       *float64
    Historical *domain.FxRate
}

// pickRate is the rate a snapshot day is converted with (DD-015 §2.4):
//  1. a rate already stored for the day and not flagged is kept, so a
//     re-run never overwrites a good rate;
//  2. a weekday with a live quote (current day) or a historical close
//     (past day) uses it;
//  3. otherwise the prior day's rate is carried with its own fx_date —
//     flagged when a weekday's fetch failed, inheriting the prior flag on a
//     weekend;
//  4. with nothing to carry the rate is unknown and flagged, never guessed.
func pickRate(date time.Time, sources rateSources, existing, prior *domain.OverallDay) domain.FxRate {
    if existing != nil && existing.EurInr != nil && !existing.FxCarried {
        return domain.FxRate{EurInr: existing.EurInr, FxDate: existing.FxDate}
    }
    weekend := isWeekend(date)
    if !weekend && sources.Live != nil {
        return domain.FxRate{EurInr: sources.Live, FxDate: domain.UTCDate(date).Format("2006-01-02")}
    }
    if sources.Historical != nil && sources.Historical.EurInr != nil {
        return *sources.Historical
    }
    if prior != nil && prior.EurInr != nil {
        return domain.FxRate{EurInr: prior.EurInr, FxDate: prior.FxDate, Carried: prior.FxCarried || !weekend}
    }
    return domain.FxRate{Carried: true}
}

// isCurrentDay reports whether date is the trading day a run at s.Now()
// records under the job's 08:00 IST cut-over (SnapshotTradingDay). Only that
// day may use the live quote; any earlier day — including yesterday once the
// cut-over has passed — uses its historical close.
func (s *SnapshotService) isCurrentDay(date time.Time) bool {
    return domain.UTCDate(date).Equal(SnapshotTradingDay(s.Now()))

// rateSourcesFor fetches the run's rate once: today's live quote for the
// current day, or the requested day's historical close for a past day.
// A failed fetch leaves the source nil (logged) and the rate carries.
func (s *SnapshotService) rateSourcesFor(ctx context.Context, logger *zap.Logger, date time.Time) rateSources {
    if s.Overall == nil || isWeekend(date) {
        return rateSources{}
    }
    if s.isCurrentDay(date) {
        rate, _, err := s.prices.GetPrice(ctx, eurInrSymbol)
        if err != nil || rate <= 0 {
            logger.Warn("snapshot: EUR→INR fetch failed; carrying the prior rate", zap.Float64("rate", rate), zap.Error(err))
            return rateSources{}
        }
        return rateSources{Live: &rate}
    }
    if s.Closes == nil {
        return rateSources{}
    }
    day := domain.UTCDate(date)
    closes, err := s.Closes.GetDailyCloses(ctx, eurInrSymbol, day.AddDate(0, 0, -7), day)
    if err != nil {
        logger.Warn("snapshot: historical EUR→INR fetch failed; carrying the prior rate", zap.Error(err))
        return rateSources{}
    }
    historical := rateOn(sortedDays(closes), closes, day.Format("2006-01-02"))
    return rateSources{Historical: &historical}
}

// rateFor resolves one user's rate for date from the day's stored document,
// the run's sources and the prior day for carry-forward.
func (s *SnapshotService) rateFor(ctx context.Context, userId primitive.ObjectID, date time.Time, sources rateSources) (domain.FxRate, error) {
    var existing, prior *domain.OverallDay
    stored, err := s.Overall.Get(ctx, userId, date)
    switch {
    case err == nil:
        existing = &stored
    case !errors.Is(err, persistence.ErrNotFound):
        return domain.FxRate{}, err
    }
    previous, err := s.Overall.LatestBefore(ctx, userId, date)
    switch {
    case err == nil:
        prior = &previous
    case !errors.Is(err, persistence.ErrNotFound):
        return domain.FxRate{}, err
    }
    return pickRate(date, sources, existing, prior), nil
}

        if point.MissingRate || point.GoldUnvalued || gold.Unavailable {
            points = append(points, point)
            continue
        }
        investedInr, currentInr := *day.StocksInvestedInr, *day.StocksCurrentInr
        if valued {
            investedInr += goldPoint.Invested
            currentInr += goldPoint.Current
        }
        investedInr, currentInr = round(investedInr), round(currentInr)
        point.InvestedInr, point.CurrentInr = &investedInr, &currentInr
        if day.EurInr != nil && *day.EurInr > 0 {
            investedEur, currentEur := round(investedInr / *day.EurInr), round(currentInr / *day.EurInr)
            point.InvestedEur, point.CurrentEur = &investedEur, &currentEur
        }
        points = append(points, point)
    }
    return points
}

// overallGold reads the gold side for the days: the overlay History uses
// plus whether gold was held each day (to tell "no price yet" from "no gold
// yet"). Unavailable is set for a gold-enabled caller whenever gold cannot
// be read — Postgres not configured or a read error — so no total appears
// without its gold.
func (h *Controller) overallGold(ctx context.Context, userId primitive.ObjectID, days []domain.OverallDay) services.OverallGold {
    caller, ok := auth.UserFromContext(ctx)
    if !ok || !caller.GoldEnabled || len(days) == 0 {
        return services.OverallGold{}
    }
    unavailable := services.OverallGold{Unavailable: true}
    if h.gold == nil {
        return unavailable
    }
    dates := make([]string, 0, len(days))
    times := make([]time.Time, 0, len(days))
    for _, day := range days {
        dates = append(dates, day.Date.UTC().Format("2006-01-02"))
        times = append(times, day.Date)
    }
    logger := h.reqLog(ctx)
    overlay, err := h.gold.HistoryOverlay(ctx, userId.Hex(), dates)
    if err != nil {
        logger.Warn("overall gold overlay failed", zap.Error(err))
        return unavailable
    }
    heldPerDay, err := h.gold.HeldOn(ctx, userId.Hex(), times)
    if err != nil {
        logger.Warn("overall gold holdings read failed", zap.Error(err))
        return unavailable
    }
    held := make(map[string]bool, len(dates))
    for i, date := range dates {
        held[date] = heldPerDay[i]
    }
    return services.OverallGold{Overlay: overlay, Held: held}
}

package services

import (

// OverallBackfill builds portfolio_overall from stored snapshots (PD-059,
// DD-015 §2.6). It only reads portfolio_snapshots. Apply fills days without
// a rate and never overwrites a rated one; Rebuild recomputes every day's
// totals and keeps a stored rate.
type OverallBackfill struct {
    snapshots *persistence.SnapshotStore
    overall   *persistence.OverallStore
    closer    DailyCloser
    logger    *zap.Logger
    Now       func() time.Time

    closes     map[string]float64
    closesFrom time.Time
    days       []string // sorted keys of closes
}

// NewOverallBackfill wires an OverallBackfill.
func NewOverallBackfill(snapshots *persistence.SnapshotStore, overall *persistence.OverallStore, closer DailyCloser, logger *zap.Logger) *OverallBackfill {
    return &OverallBackfill{snapshots: snapshots, overall: overall, closer: closer, logger: logger, Now: func() time.Time { return time.Now().UTC() }}
}

// OverallBackfillOptions selects the mode.
type OverallBackfillOptions struct {
    DryRun  bool
    Rebuild bool
}

// OverallBackfillReport is one user's outcome. The dry run fills it with
// exactly the selection apply would make, so it is a true preview:
// SkippedRated is what apply leaves alone, ToWrite what it would write,
// WithoutRate the ToWrite rows it cannot convert, and Min/MaxCurrent span
// only the ToWrite rows.
type OverallBackfillReport struct {
    Rows         int
    SkippedRated int
    ToWrite      int
    Written      int
    WithoutRate  int
    MinCurrent   float64
    MaxCurrent   float64
}

// RunUser backfills one user. Both stores are read first; Yahoo is called
// only when some row needs a rate it does not already have, so a rebuild
// with every rate stored works through a Yahoo outage.
func (b *OverallBackfill) RunUser(ctx context.Context, userId primitive.ObjectID, opts OverallBackfillOptions) (_ OverallBackfillReport, retErr error) {
    ctx, span := tracer.Start(ctx, "OverallBackfill.RunUser")
    defer endSpan(span, &retErr)
    report := OverallBackfillReport{}
    today := domain.UTCDate(b.Now())
    rows, err := b.snapshots.List(ctx, userId, time.Time{}, today)
    if err != nil || len(rows) == 0 {
        return report, err
    }
    earliest := rows[len(rows)-1].Date // List is newest first
    existing, err := b.overall.List(ctx, userId, earliest, today)
    if err != nil {
        return report, err
    }
    rated := make(map[string]domain.OverallDay, len(existing))
    for _, day := range existing {
        if day.EurInr != nil {
            rated[day.Date.UTC().Format("2006-01-02")] = day
        }
    }

    // Selection, shared by dry run, apply and rebuild: apply skips rated
    // days; rebuild keeps their stored rate. Only the rest need Yahoo.
    var needRateFrom *time.Time
    for _, row := range rows {
        if _, ok := rated[row.Date.UTC().Format("2006-01-02")]; !ok {
            date := row.Date
            needRateFrom = &date // rows are newest first, so this ends earliest
        }
    }
    if needRateFrom != nil {
        if err := b.ensureCloses(ctx, *needRateFrom, today); err != nil {
            return report, err
        }
    }

    report.MinCurrent = -1
    for _, row := range rows {
        report.Rows++
        date := row.Date.UTC().Format("2006-01-02")
        stored, isRated := rated[date]
        if isRated && !opts.Rebuild {
            report.SkippedRated++
            continue
        }
        rate := domain.FxRate{EurInr: stored.EurInr, FxDate: stored.FxDate, Carried: stored.FxCarried}
        if !isRated {
            rate = rateOn(b.days, b.closes, date)
        }
        report.ToWrite++
        if _, current := domain.OverallFromBuckets(row.Buckets, rate.EurInr); current == nil {
            report.WithoutRate++
        } else {
            if report.MinCurrent < 0 || *current < report.MinCurrent {
                report.MinCurrent = *current
            }
            report.MaxCurrent = max(report.MaxCurrent, *current)
        }
        if opts.DryRun {
            continue
        }
        written := true
        if opts.Rebuild {
            err = b.overall.Put(ctx, userId, row.Date, rate, row.Buckets)
        } else {
            // FillUnrated re-checks "no rate" in the database, a guard
            // against a live write landing between the read and this one.
            written, err = b.overall.FillUnrated(ctx, userId, row.Date, rate, row.Buckets)
        }
        if err != nil {
            return report, err
        }
        if written {
            report.Written++
        }
    }
    if report.MinCurrent < 0 {
        report.MinCurrent = 0
    }
    return report, nil
}

// ensureCloses fetches EUR→INR closes once per run, refetching only when a
// user's history starts earlier than the series already held.
func (b *OverallBackfill) ensureCloses(ctx context.Context, from, to time.Time) error {
    if b.closes != nil && !from.Before(b.closesFrom) {
        return nil
    }
    closes, err := b.closer.GetDailyCloses(ctx, eurInrSymbol, from.AddDate(0, 0, -7), to)
    if err != nil {
        return err
    }
    b.closes, b.closesFrom, b.days = closes, from, sortedDays(closes)
    return nil
}

func runMigrateOverall(cmd *cobra.Command, _ []string) error {
    ctx, store, logger, closeSession, err := cliSession(cmd)
    if err != nil {
        return err
    }
    defer closeSession()

// overallBackfillUsers is every user by _id, disabled ones included (their
// history is kept), or the one --user names.
func overallBackfillUsers(ctx context.Context, store *persistence.Store, only string) ([]primitive.ObjectID, error) {
    if only != "" {
        userId, err := primitive.ObjectIDFromHex(only)
        if err != nil {
            return nil, fmt.Errorf("parse --user %q: %w", only, err)
        }
        return []primitive.ObjectID{userId}, nil
    }
    users, err := store.Users.ListAllByID(ctx)
    if err != nil {
        return nil, err
    }
    userIds := make([]primitive.ObjectID, 0, len(users))
    for _, user := range users {
        userIds = append(userIds, user.ID)
    }
    return userIds, nil
}

        zap.Int("to_write", report.ToWrite),
        zap.Int("written", report.Written),
        zap.Int("without_rate", report.WithoutRate),
        zap.Float64("min_current_inr", report.MinCurrent),
        zap.Float64("max_current_inr", report.MaxCurrent),
        zap.String("currency", domain.CurrencyINR),
    )

package cmd

import (
    "context"
    "slices"

    "go.mongodb.org/mongo-driver/bson/primitive"
    "go.uber.org/zap"

    "portfolio-dashboard/internal/persistence"
    "portfolio-dashboard/internal/services"
)

// overallBackfillStep is the tracked form of `migrate overall` (PD-059):
// one user per batch, the cursor is the last user id done, so a killed run
// resumes with the next user. Inserting only missing days keeps every batch
// idempotent.
type overallBackfillStep struct {
    store    *persistence.Store
    backfill *services.OverallBackfill
    logger   *zap.Logger
}

func newOverallBackfillStep(store *persistence.Store, logger *zap.Logger) overallBackfillStep {
    backfill := services.NewOverallBackfill(store.Snapshots, store.Overall, services.NewPriceService(logger), logger)
    return overallBackfillStep{store: store, backfill: backfill, logger: logger}
}

func (overallBackfillStep) Version() string { return "20261118_02" }

func (s overallBackfillStep) Run(ctx context.Context, cursor *string) (*string, error) {
    userIds, err := overallBackfillUsers(ctx, s.store, "")
    if err != nil {
        return nil, err
    }
    next, ok := nextOverallUser(userIds, cursor)
    if !ok {
        return nil, nil
    }
    opts := services.OverallBackfillOptions{}
    report, err := s.backfill.RunUser(ctx, next, opts)
    logOverallBackfillReport(s.logger, next, opts, report)
    if err != nil {
        return nil, err
    }
    done := next.Hex()
    return &done, nil
}

// nextOverallUser is the first user after cursor in _id order.
func nextOverallUser(userIds []primitive.ObjectID, cursor *string) (primitive.ObjectID, bool) {
    start := 0
    if cursor != nil {
        start = slices.IndexFunc(userIds, func(userId primitive.ObjectID) bool { return userId.Hex() > *cursor })
        if start < 0 {
            return primitive.ObjectID{}, false
        }
    }
    if start >= len(userIds) {
        return primitive.ObjectID{}, false
    }
    return userIds[start], true
}
